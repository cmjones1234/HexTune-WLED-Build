      - name: Run PlatformIO

        run: |

          rm -rf /tmp/WLEDtemp/* || true

          mkdir -p /tmp/WLEDtemp

          wget \
            https://api.github.com/repos/wled/WLED/zipball/nightly \
            -O /tmp/WLEDtemp/mm.zip

          unzip \
            /tmp/WLEDtemp/mm.zip \
            -d /tmp/WLEDtemp

          find \
            /tmp/WLEDtemp \
            -maxdepth 1 \
            -mindepth 1 \
            -name "*" \
            -type d \
            >> /tmp/WLEDtemp/wledfoldername

          cd \
            `cat /tmp/WLEDtemp/wledfoldername`

          npm install

          echo '[env:custom_build]' \
            >> platformio.ini

          echo 'extends = env:esp32s3_4M_qspi' \
            >> platformio.ini

          echo 'build_flags = ${env:esp32s3_4M_qspi.build_flags} -D DATA_PINS=5 -D PIXEL_COUNTS=7' \
            >> platformio.ini

          echo 'custom_usermods = audioreactive' \
            >> platformio.ini

          echo 'build_unflags = -D WLED_RELEASE_NAME' \
            >> platformio.ini

          pio run -e custom_build || \
          pio run -e custom_build
